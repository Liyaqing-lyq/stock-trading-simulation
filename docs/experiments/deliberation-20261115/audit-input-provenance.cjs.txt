const fs=require('node:fs');const path=require('node:path');const crypto=require('node:crypto');const {isDeepStrictEqual}=require('node:util');const assert=require('node:assert/strict');
const root=fs.realpathSync(path.resolve(__dirname,'../..'));const inside=p=>{const q=fs.realpathSync(path.resolve(p));assert(q.toLowerCase().startsWith(root.toLowerCase()+path.sep),'outside project');return q;};
const original=inside(process.argv[2]),auditDir=inside(process.argv[3]);const audit=JSON.parse(fs.readFileSync(path.join(auditDir,'audit.json'),'utf8'));const config=JSON.parse(fs.readFileSync(path.join(original,'config.json'),'utf8'));const result=JSON.parse(fs.readFileSync(path.join(original,'result.json'),'utf8'));const decisions=new Map(audit.decisions.state.decisions.map(d=>[d.actionId,d]));let checked=0,replies=0;const perModel={};
for(const name of fs.readdirSync(original).filter(n=>/^call-\d+\.json$/.test(n)).sort((a,b)=>+a.match(/\d+/)[0]-+b.match(/\d+/)[0])){const call=JSON.parse(fs.readFileSync(path.join(original,name),'utf8'));const d=decisions.get(call.actionId);assert(d,'missing decision '+name);assert.equal(call.runId,d.runId);assert.equal(call.inputText,d.inputText,'input differs '+name);const digest=crypto.createHash('sha256').update(JSON.stringify({domain:'decision-input-v1',toolContractVersion:d.toolContractVersion,allowedTools:d.allowedTools,inputText:call.inputText}),'utf8').digest('hex');assert.equal(digest,d.inputDigest);assert.equal(Buffer.byteLength(call.inputText,'utf8'),d.plan.inputBytes);assert(isDeepStrictEqual(call.allowedTools,d.allowedTools),'tools differ');assert(isDeepStrictEqual(call.plan,d.plan),'plan differs');const a=d.attempts.find(a=>a.attemptId===call.attemptId);assert(a,'missing attempt');assert.equal(a.inputDigest,digest);assert.equal(a.attemptNo,call.attemptNo);assert.equal(a.recoveryGeneration,call.recoveryGeneration);const raw=JSON.parse(fs.readFileSync(path.join(original,name.replace('call-','reply-')),'utf8'));assert(raw.reply&&!raw.failureCode,'unexpected provider failure branch');assert.equal(a.replyBody,raw.reply.body,'reply differs '+name);assert.equal(a.replyReceived,true);assert(isDeepStrictEqual(a.declaredUsage,raw.reply.usage),'usage differs '+name);checked++;replies++;const model=call.plan.route.model;perModel[model]=(perModel[model]||0)+1;}
const sha=p=>crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex');const sourceHash=sha(path.join(original,'world.journal'));assert.equal(sourceHash,sha(path.join(auditDir,'world.journal')),'journal copy differs');assert.equal(audit.position,2075);assert.equal(audit.timeMillis,5105000);assert.equal(Number(result.physicalInFlight??result.live?.physicalInFlight),0);const out={scope:'Independent original saved provider calls/replies compared byte-for-byte with recovered authoritative decisions; no model calls; applies to this audit candidate only',sourceCommit:config.sourceCommit,sourceJournalSha256:sourceHash,originalConfigSha256:sha(path.join(original,'config.json')),originalSystemPromptSha256:sha(path.join(original,'system-prompt.txt')),position:audit.position,timeMillis:audit.timeMillis,callsChecked:checked,replyBodiesAndDeclaredUsageChecked:replies,inputAndDigestAndPlanAndToolsAndAttemptIdentityMatch:true,requestedModels:perModel,providerReportedModelVersion:'UNKNOWN (not retained by this older provider adapter)',httpCalls:0};fs.writeFileSync(path.join(auditDir,'original-call-provenance.json'),JSON.stringify(out,null,2),{flag:'wx'});console.log(JSON.stringify(out));
